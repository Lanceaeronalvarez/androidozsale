PROJECT_DIR=$(dirname $(dirname "$0"))/
OUTPUT_DIR="$PROJECT_DIR"apkbuilds/

error=0

flag_error() {
     if [ $error -eq 0 ]; then
          if [ $1 -ne 0 ]; then               
              error=1
          fi
     fi
}

print_green(){
    printf "\e[1;32m$1\e[0m"
}

print_blue(){
    printf "\e[1;34m$1\e[0m"
}

print_red(){
    printf "\e[1;31m$1\e[0m"
}

print_yellow(){
    printf "\e[1;33m$1\e[0m"
}

copy_to_output_folder(){
	if [ ! -d $OUTPUT_DIR ]
	then
		mkdir $OUTPUT_DIR
	fi
	cp -f "$PROJECT_DIR"app/build/outputs/apk/"$flavorName"/release/app-"$flavorName"-release.apk "$OUTPUT_DIR"
	flag_error $?
	if [ -f "$OUTPUT_DIR"app-"$flavorName"-release.apk ]
	then
		print_blue "\n\n\n Finished copying APK to output directory\n"
	else
		print_yellow "\n\n\n APK is not be copied into output directory\n"
	fi
}
